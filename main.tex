% Integrate complete immutable floats without nesting or editing their assets.
% Both kinds of inherited tables use ACL's full text width; only tabular
% content is scaled by the archived files themselves, never the captions.
\newenvironment{archivedtables}[1][!t]{%
  \expandafter\let\expandafter\archivedwide\csname table*\endcsname
  \expandafter\let\expandafter\endarchivedwide\csname endtable*\endcsname
  \let\archivedcaption\caption
  \let\archivedlabel\label
  \renewenvironment{table}[1][]{%
    \archivedwide[#1]%
    \renewcommand{\caption}[1]{\def\archivedcaptiontext{####1}}%
    \renewcommand{\label}[1]{\def\archivedlabeltext{####1}}%
  }{%
    \par\archivedcaption{\archivedcaptiontext}%
    \archivedlabel{\archivedlabeltext}\endarchivedwide}%
}{}
% Keep diagnostic pairs together without delaying later numbered figures.

}{\end{figure*}}
\newenvironment{widearchivedfigures}{%
  \expandafter\let\expandafter\archivedwidefigure\csname figure*\endcsname
  \expandafter\let\expandafter\endarchivedwidefigure\csname endfigure*\endcsname

}{}
